/** Contracts of the additive founder mission service. No customer authority is inherited. */
export type Json = null | boolean | number | string | Json[] | { [key: string]: Json };
export type JsonObject = { [key: string]: Json };
export type MissionMode = "prepare" | "live" | "test";
export type Risk = "read" | "prepare" | "write" | "financial" | "publish" | "code";

export interface AuthenticatedActor {
  ownerId: string;
  channel: "telegram" | "admin";
  subject: string;
  assurance: "owner" | "mfa";
  authenticatedAt: string;
}

export interface Evidence {
  kind: string;
  source: string;
  observedAt: string;
  reference: string;
  accountId?: string;
  data: JsonObject;
}
export interface Need {
  id: string;
  kind: "access" | "approval" | "information" | "capability" | "verification";
  service: string;
  description: string;
  accountId?: string;
  permissions?: string[];
  officialUrl?: string;
}
export interface ActionResult {
  summary: string;
  output: JsonObject;
  evidence: Evidence[];
}
export interface ExecutionContext {
  missionId: string;
  stepId: string;
  ownerId: string;
  mode: MissionMode;
  signal: AbortSignal;
  idempotencyKey: string;
  now: () => Date;
}
export interface Capability {
  id: string;
  version: number;
  risk: Risk;
  description: string;
  validate(input: unknown): JsonObject;
  prerequisites(context: ExecutionContext, input: JsonObject): Promise<Need[]>;
  execute(context: ExecutionContext, input: JsonObject): Promise<ActionResult>;
  /** A prior uncertain effect must be read back, never blindly retried. */
  reconcile?(context: ExecutionContext, input: JsonObject): Promise<ActionResult | null>;
}

/** Produced by trusted server policy, never accepted from a natural-language plan. */
export interface Mandate {
  id: string;
  expiresAt: string;
  capabilityIds: string[];
  /** Legacy metadata retained for immutable missions. Effects require the separate
   * durable approval ledger; these hashes never authorize a new dispatch. */
  approvedInputHashes: string[];
  assurance: "owner" | "mfa";
}
export type StepStatus = "pending" | "executing" | "waiting" | "succeeded" | "failed" | "uncertain";
export interface MissionStep {
  id: string;
  capabilityId: string;
  capabilityVersion: number;
  input: JsonObject;
  dependsOn: string[];
  status: StepStatus;
  attempts: number;
  startedAt?: string;
  result?: ActionResult;
  needs?: Need[];
  error?: string;
}
export interface Mission {
  id: string;
  ownerId: string;
  requestId: string;
  requestHash: string;
  objective: string;
  actor: AuthenticatedActor;
  mode: MissionMode;
  planVersion: number;
  status: "queued" | "running" | "waiting" | "succeeded" | "failed" | "cancelled";
  createdAt: string;
  updatedAt: string;
  mandate: Mandate;
  steps: MissionStep[];
}
export interface Lease { holder: string; generation: number; expiresAt: string; }
export interface StoredMission {
  mission: Mission;
  revision: number;
  control: "run" | "pause" | "cancel";
  lease: Lease | null;
}
export interface MissionEvent {
  type: string;
  at: string;
  stepId?: string;
  data: JsonObject;
}
export type MissionCommand =
  | { type: "pause" | "cancel" | "continue" }
  | { type: "recheck"; stepId: string; needId: string; planVersion: number }
  /** Internal only: generated by authenticated server policy, never parsed from chat. */
  | { type: "renew_read_mandate"; mandate: Mandate; planVersion: number };

export interface MissionRepository {
  /** Atomic insert or replay by ownerId/requestId. Different hash must reject. */
  create(mission: Mission): Promise<{ record: StoredMission; replay: boolean }>;
  get(ownerId: string, missionId: string): Promise<StoredMission | null>;
  /** Exact owner/request lookup for transport replay before planning; never a list scan. */
  getByRequestId(ownerId: string, requestId: string): Promise<StoredMission | null>;
  list(ownerId: string, limit: number): Promise<StoredMission[]>;
  /** Claim one runnable mission. Uses the database clock and fences old holders. */
  claim(holder: string, leaseSeconds: number): Promise<StoredMission | null>;
  /** Save under the current unexpired lease and expected revision; append event atomically. */
  commit(record: StoredMission, mission: Mission, event: MissionEvent, release: boolean): Promise<StoredMission>;
  /** Commands change control separately from worker state; recheck never accepts proof from chat. */
  command(ownerId: string, missionId: string, command: MissionCommand): Promise<StoredMission>;
  events(ownerId: string, missionId: string, limit: number): Promise<MissionEvent[]>;
  consumeNonce(nonce: string, expiresAt: string): Promise<boolean>;
}
