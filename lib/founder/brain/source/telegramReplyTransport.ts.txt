import { containsSecret } from "./model.js";

export interface TelegramReplyRequest {
  ownerId: string; botScopeId: string; telegramSubject: string;
  replyToMessageId: number; text: string;
}
export interface TelegramReplyReceipt {
  messageId: number; chatId: string; botId: string; text: string; observedAt: string;
}
export class TelegramReplyTransportError extends Error {
  constructor(public readonly code: "configuration" | "refused" | "rejected" | "unknown") {
    super(`telegram_reply_${code}`); this.name = "TelegramReplyTransportError";
  }
}
const positive = (v: unknown): v is number => Number.isSafeInteger(v) && Number(v) > 0;
const object = (v: unknown): v is Record<string, unknown> => !!v && typeof v === "object" && !Array.isArray(v);
const identity = (v: string) => /^[1-9][0-9]{0,15}$/.test(v) && BigInt(v) <= 4503599627370495n;
const proof = /\bv1\.[A-Za-z0-9_-]{20,}\.[A-Za-z0-9_-]{43}\b/;

/** One attempt only. The caller must durably record dispatch BEFORE send().
 * No bot registration, polling, payment, fallback sender or provider retry. */
export function createTelegramReplyTransport(options: {
  ownerId: string; botScopeId: string; telegramSubject: string; botToken: string;
  forbiddenSecrets: string[]; enabled: () => boolean; transport: typeof fetch;
  currentOwnerAccess: () => Promise<boolean>; timeoutMs?: number; now?: () => Date;
}) {
  const { ownerId, botScopeId, telegramSubject, botToken } = options;
  const timeoutMs = options.timeoutMs ?? 8_000, now = options.now ?? (() => new Date());
  const match = /^([1-9][0-9]{0,15}):([A-Za-z0-9_-]{32,128})$/.exec(botToken);
  if (!ownerId || ownerId.length > 128 || !/^[A-Za-z0-9_-]{1,64}$/.test(botScopeId)
    || !identity(telegramSubject) || !match || !identity(match[1]!) || new Set(match[2]).size < 8
    || options.forbiddenSecrets.includes(botToken) || typeof options.currentOwnerAccess !== "function"
    || !Number.isInteger(timeoutMs) || timeoutMs < 50 || timeoutMs > 15_000) {
    throw new TelegramReplyTransportError("configuration");
  }
  const botId = match[1]!, secrets = [botToken, ...options.forbiddenSecrets].filter(Boolean);
  const url = `https://api.telegram.org/bot${botToken}/sendMessage`;
  const enabled = () => { try { return options.enabled() === true; } catch { return false; } };
  return {
    async send(value: TelegramReplyRequest, parentSignal: AbortSignal): Promise<TelegramReplyReceipt> {
      const input = structuredClone(value);
      if (!enabled() || parentSignal.aborted || !object(input)
        || Object.keys(input).sort().join(",") !== "botScopeId,ownerId,replyToMessageId,telegramSubject,text"
        || input.ownerId !== ownerId || input.botScopeId !== botScopeId || input.telegramSubject !== telegramSubject
        || !positive(input.replyToMessageId) || typeof input.text !== "string" || !input.text.trim()
        || input.text.length > 3_500 || Buffer.byteLength(input.text, "utf8") > 14_000
        || containsSecret(input.text) || proof.test(input.text) || secrets.some(s => input.text.includes(s))) {
        throw new TelegramReplyTransportError("refused");
      }
      const controller = new AbortController();
      const signal = AbortSignal.any([parentSignal, controller.signal]);
      let dispatched = false, timer: ReturnType<typeof setTimeout> | undefined;
      let onAbort: () => void = () => {};
      const interrupted = new Promise<never>((_, reject) => {
        onAbort = () => reject(new TelegramReplyTransportError(dispatched ? "unknown" : "refused"));
        signal.addEventListener("abort", onAbort, { once: true });
        timer = setTimeout(() => controller.abort(), timeoutMs);
      });
      try {
        return await Promise.race([(async () => {
          let allowed = false;
          try { allowed = await options.currentOwnerAccess() === true; } catch { /* No identity fallback. */ }
          if (!allowed || !enabled() || signal.aborted) throw new TelegramReplyTransportError("refused");
          const body = JSON.stringify({ chat_id: telegramSubject, text: input.text,
            reply_parameters: { message_id: input.replyToMessageId, allow_sending_without_reply: false },
            link_preview_options: { is_disabled: true }, protect_content: true, allow_paid_broadcast: false });
          dispatched = true;
          let response: Response;
          try { response = await options.transport(url, { method: "POST", redirect: "error",
            headers: { "Content-Type": "application/json" }, body, signal }); }
          catch { throw new TelegramReplyTransportError("unknown"); }
          if (signal.aborted || !enabled() || response.redirected
            || !/^application\/json(?:\s*;|$)/i.test(response.headers.get("content-type") ?? "")
            || (response.url && response.url !== url) || !response.body) throw new TelegramReplyTransportError("unknown");
          const declared = response.headers.get("content-length");
          if (declared && (!/^\d+$/.test(declared) || Number(declared) > 65_536)) throw new TelegramReplyTransportError("unknown");
          const reader = response.body.getReader(), chunks: Uint8Array[] = [];
          let size = 0, decoded: unknown;
          const cancel = () => { void reader.cancel().catch(() => {}); };
          signal.addEventListener("abort", cancel, { once: true });
          try {
            for (;;) {
              const part = await reader.read();
              if (signal.aborted) throw new Error();
              if (part.done) break;
              size += part.value.byteLength;
              if (size > 65_536) throw new Error();
              chunks.push(part.value);
            }
            decoded = JSON.parse(new TextDecoder("utf-8", { fatal: true }).decode(Buffer.concat(chunks)));
          } catch { throw new TelegramReplyTransportError("unknown"); }
          finally { signal.removeEventListener("abort", cancel); cancel(); }
          if (!enabled() || signal.aborted || !object(decoded)) throw new TelegramReplyTransportError("unknown");
          // Explicit API refusals are not delivery receipts. Never copy description,
          // parameters or supplier URLs into a user message or an exception.
          if (decoded.ok === false && [400, 401, 403, 404, 429].includes(response.status)
            && decoded.error_code === response.status && !Object.hasOwn(decoded, "result")) throw new TelegramReplyTransportError("rejected");
          const result = decoded.result;
          if (response.status !== 200 || decoded.ok !== true || !object(result) || !positive(result.message_id)
            || !object(result.chat) || result.chat.type !== "private" || !positive(result.chat.id)
            || String(result.chat.id) !== telegramSubject || !object(result.from) || result.from.is_bot !== true
            || !positive(result.from.id) || String(result.from.id) !== botId || result.text !== input.text
            || !object(result.reply_to_message) || result.reply_to_message.message_id !== input.replyToMessageId) {
            throw new TelegramReplyTransportError("unknown");
          }
          return { messageId: result.message_id, chatId: telegramSubject, botId, text: input.text,
            observedAt: now().toISOString() };
        })(), interrupted]);
      } catch (error) {
        if (error instanceof TelegramReplyTransportError) throw error;
        throw new TelegramReplyTransportError(dispatched ? "unknown" : "refused");
      } finally { clearTimeout(timer); signal.removeEventListener("abort", onAbort); controller.abort(); }
    },
  };
}
