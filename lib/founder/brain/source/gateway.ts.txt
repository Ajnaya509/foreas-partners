const BASE = "/founder/v1/missions";
const MAX_RESPONSE_BYTES = 4 * 1_048_576;
const MAX_BODY_BYTES = 24 * 1_024;
const ID = /^[A-Za-z0-9][A-Za-z0-9_-]{0,127}$/;
const REQUEST_ID = /^[A-Za-z0-9_:.-]{1,160}$/;
const STEP_ID = /^[a-z][a-z0-9_-]{0,63}$/;
const INTENT_ID = /^[a-f0-9]{8}(?:-[a-f0-9]{4}){3}-[a-f0-9]{12}$/;
const DIGEST = /^[a-f0-9]{64}$/;
const MAX_ACTIONS = 100;
const MAX_MANIFEST_BYTES = 24 * 1_024;
const PROOF = /\bv1\.[A-Za-z0-9_-]{20,}\.[A-Za-z0-9_-]{43}\b/;
const STATUS = new Set(["queued", "running", "waiting", "succeeded", "failed", "cancelled"]);
const STEP_STATUS = new Set(["pending", "executing", "waiting", "succeeded", "failed", "uncertain"]);

export type GatewayErrorCode = "configuration" | "request_invalid" | "authority_refused" | "timeout"
  | "transport_unavailable" | "service_refused" | "response_invalid" | "outcome_unknown";
export class MissionGatewayError extends Error {
  constructor(public readonly code: GatewayErrorCode, public readonly status?: number) {
    super(`mission_gateway_${code}`);
    this.name = "MissionGatewayError";
  }
}
const fail = (code: GatewayErrorCode, status?: number): never => { throw new MissionGatewayError(code, status); };
const object = (value: unknown): value is Record<string, unknown> => value !== null && typeof value === "object" && !Array.isArray(value);

export interface GatewayMissionView extends JsonObject {
  id: string;
  objective: string;
  status: "queued" | "running" | "waiting" | "succeeded" | "failed" | "cancelled";
  control: "run" | "pause" | "cancel";
  mode: "prepare" | "live" | "test";
  steps: JsonObject[];
}
export interface GatewayMissionResponse { mission: GatewayMissionView; replay?: boolean; }
export interface GatewayMissionList { missions: GatewayMissionView[]; coverage: { limit: number; exhaustive: boolean }; }
/** Admin projection only. Never pass this object to a Telegram response renderer. */
export interface GatewayActionView {
  id: string; missionId: string; stepId: string; planVersion: number;
  manifest: JsonObject; manifestHash: string; expiresAt: string; revokedAt: string | null;
  approval: null | { approvedAt: string; expiresAt: string; revokedAt: string | null; consumedAt: string | null };
}
export interface GatewayActionList { actions: GatewayActionView[]; }
export interface GatewayActionResponse { action: GatewayActionView; }
export interface GatewayActionApprovalResponse extends GatewayActionResponse { execution: "not_confirmed"; }
/** Deliberately excludes server-generated mandates and approvals. */
export type GatewayMissionCommand = { type: "pause" | "cancel" | "continue" }
  | { type: "recheck"; stepId: string; needId: string; planVersion: number };

export interface MissionGatewayOptions {
  /** Trusted server configuration only, HTTPS origin with no path/query/credentials. */
  brainOrigin: string;
  authority: Omit<MissionAuthorityOptions, "consumeNonce">;
  /** Revalidates server session/MFA or verified receiver context on every attempt. */
  authenticate: () => Promise<AuthenticatedActor>;
  transport: typeof fetch;
  /** Entire logical call budget, including authentication, response and retry. */
  timeoutMs?: number;
  /** Only GET and create may retry automatically. Commands never do. */
  attempts?: 1 | 2;
}

function originFromConfiguration(value: string): string {
  let url: URL;
  try { url = new URL(value); } catch { return fail("configuration"); }
  if (url.protocol !== "https:" || url.username || url.password || url.search || url.hash
    || url.pathname !== "/" || ![url.origin, `${url.origin}/`].includes(value) || !url.hostname) return fail("configuration");
  return url.origin;
}
function commandBody(value: unknown): GatewayMissionCommand {
  if (!object(value)) return fail("request_invalid");
  if ((value.type === "pause" || value.type === "cancel" || value.type === "continue") && Object.keys(value).length === 1) {
    return { type: value.type };
  }
  if (value.type === "recheck" && Object.keys(value).sort().join(",") === "needId,planVersion,stepId,type"
    && typeof value.stepId === "string" && /^[a-z][a-z0-9_-]{0,63}$/.test(value.stepId)
    && typeof value.needId === "string" && /^[A-Za-z0-9_.:-]{1,128}$/.test(value.needId)
    && Number.isSafeInteger(value.planVersion) && Number(value.planVersion) > 0) return value as unknown as GatewayMissionCommand;
  return fail("request_invalid");
}
function missionId(value: unknown): string {
  if (typeof value !== "string" || !ID.test(value)) return fail("request_invalid");
  return value;
}
function exactKeys(value: unknown, keys: string): value is Record<string, unknown> {
  return object(value) && Object.keys(value).sort().join(",") === keys;
}
function requestField(value: unknown, pattern: RegExp): string {
  if (typeof value !== "string" || !pattern.test(value)) return fail("request_invalid");
  return value;
}
const positive = (value: unknown): value is number => Number.isSafeInteger(value) && Number(value) > 0;
function date(value: unknown): value is string {
  if (typeof value !== "string" || !/^\d{4}-(?:0[1-9]|1[0-2])-(?:0[1-9]|[12]\d|3[01])T(?:[01]\d|2[0-3]):[0-5]\d:[0-5]\d(?:\.\d{1,6})?(?:Z|[+-](?:[01]\d|2[0-3]):[0-5]\d)$/.test(value)
    || !Number.isFinite(Date.parse(value))) return false;
  const [year, month, day] = value.slice(0, 10).split("-").map(Number);
  return day <= new Date(Date.UTC(year, month, 0)).getUTCDate();
}
const nullableDate = (value: unknown) => value === null || date(value);
function readAction(value: unknown, expected: { missionId?: string; stepId?: string; id?: string; manifestHash?: string }): GatewayActionView {
  if (!exactKeys(value, "approval,expiresAt,id,manifest,manifestHash,missionId,planVersion,revokedAt,stepId")
    || typeof value.id !== "string" || !INTENT_ID.test(value.id)
    || typeof value.missionId !== "string" || !ID.test(value.missionId)
    || typeof value.stepId !== "string" || !STEP_ID.test(value.stepId) || !positive(value.planVersion)
    || !date(value.expiresAt) || !nullableDate(value.revokedAt) || !object(value.manifest)
    || typeof value.manifestHash !== "string" || !DIGEST.test(value.manifestHash)
    || Object.entries(expected).some(([key, expectedValue]) => value[key] !== expectedValue)) return fail("response_invalid");
  const encoded = canonical(value.manifest);
  if (Buffer.byteLength(encoded) > MAX_MANIFEST_BYTES || hash(value.manifest) !== value.manifestHash) return fail("response_invalid");
  // These are transport/session credentials, never part of an actionable public description.
  // Descriptive provider URLs remain inert data; only the configured brain origin is contacted.
  function privateFields(v: unknown): boolean {
    if (Array.isArray(v)) return v.some(privateFields);
    return object(v) && Object.entries(v).some(([key, child]) => /^(?:password|secret|secretKey|apiKey|api_key|access_token|refresh_token|authorization|cookie|private_key|attestation|mfaAt|authenticatedAt|x-mission-authority)$/i.test(key) || privateFields(child));
  }
  if (privateFields(value.manifest)) return fail("response_invalid");
  const approval = value.approval;
  if (approval !== null && (!exactKeys(approval, "approvedAt,consumedAt,expiresAt,revokedAt")
    || !date(approval.approvedAt) || !date(approval.expiresAt) || !nullableDate(approval.revokedAt) || !nullableDate(approval.consumedAt)
    || Date.parse(approval.expiresAt) <= Date.parse(approval.approvedAt)
    || Date.parse(approval.expiresAt) - Date.parse(approval.approvedAt) > 300_000
    || Date.parse(approval.expiresAt) > Date.parse(value.expiresAt)
    || (approval.consumedAt !== null && (Date.parse(approval.consumedAt as string) < Date.parse(approval.approvedAt)
      || Date.parse(approval.consumedAt as string) >= Date.parse(approval.expiresAt))))) return fail("response_invalid");
  return value as unknown as GatewayActionView;
}
function readMission(value: unknown): GatewayMissionView {
  if (!object(value) || typeof value.id !== "string" || !ID.test(value.id)
    || typeof value.objective !== "string" || typeof value.status !== "string" || !STATUS.has(value.status)
    || !["run", "pause", "cancel"].includes(String(value.control)) || !["prepare", "live", "test"].includes(String(value.mode))
    || !Array.isArray(value.steps) || !value.steps.length || value.steps.length > 12
    || value.steps.some(s => !object(s) || typeof s.id !== "string" || typeof s.status !== "string" || !STEP_STATUS.has(s.status)
      || typeof s.capability !== "string" || !/^[a-z][a-z0-9_.]{2,95}$/.test(s.capability)
      || (s.status === "succeeded" && (!object(s.result) || typeof s.result.summary !== "string" || !s.result.summary.trim()
        || !object(s.result.output) || !Array.isArray(s.result.evidence) || !s.result.evidence.length)))
    || (value.status === "succeeded" && value.steps.some(s => s.status !== "succeeded"))) {
    return fail("response_invalid");
  }
  return value as unknown as GatewayMissionView;
}
async function bounded<T>(operation: (signal: AbortSignal) => Promise<T>, ms: number): Promise<T> {
  const controller = new AbortController();
  let timer: ReturnType<typeof setTimeout> | undefined;
  try {
    return await Promise.race([operation(controller.signal), new Promise<never>((_, reject) => {
      timer = setTimeout(() => { controller.abort(); reject(new MissionGatewayError("timeout")); }, ms);
    })]);
  } finally { clearTimeout(timer); controller.abort(); }
}
async function jsonResponse(response: Response): Promise<unknown> {
  if (!/^application\/json(?:\s*;|$)/i.test(response.headers.get("content-type") ?? "")) return fail("response_invalid");
  const advertised = response.headers.get("content-length");
  if (advertised && (!/^\d+$/.test(advertised) || Number(advertised) > MAX_RESPONSE_BYTES)) return fail("response_invalid");
  if (!response.body) return fail("response_invalid");
  const reader = response.body.getReader();
  const chunks: Uint8Array[] = [];
  let length = 0;
  try {
    while (true) {
      const chunk = await reader.read();
      if (chunk.done) break;
      length += chunk.value.byteLength;
      if (length > MAX_RESPONSE_BYTES) return fail("response_invalid");
      chunks.push(chunk.value);
    }
    return JSON.parse(Buffer.concat(chunks).toString("utf8"));
  } catch (error) {
    if (error instanceof MissionGatewayError) throw error;
    return fail("response_invalid");
  } finally { void reader.cancel().catch(() => {}); reader.releaseLock(); }
}

/** No Telegram, model, supplier URL, logging or production side effect is called here. */
export function createMissionGateway(options: MissionGatewayOptions) {
  const origin = originFromConfiguration(options.brainOrigin);
  const timeoutMs = options.timeoutMs ?? 10_000;
  const attempts = options.attempts ?? 2;
  if (!Number.isSafeInteger(timeoutMs) || timeoutMs < 100 || timeoutMs > 20_000
    || ![1, 2].includes(attempts) || typeof options.authenticate !== "function" || typeof options.transport !== "function"
    || !object(options.authority) || !object(options.authority.gateways)
    || !Array.isArray(options.authority.forbiddenSecrets)
    || options.authority.forbiddenSecrets.some(s => typeof s !== "string")) return fail("configuration");
  const authority = { ...options.authority,
    // Signing never consumes a nonce. Consumption belongs exclusively to the brain repository.
    consumeNonce: async (): Promise<boolean> => { throw new Error("issuer_must_not_consume_nonce"); },
  };
  const secrets = [...Object.values(authority.gateways).map(g => g?.secret ?? ""), ...authority.forbiddenSecrets].filter(Boolean);
  function safe(value: unknown): void {
    let encoded: string;
    try { encoded = canonical(value); } catch { return fail("response_invalid"); }
    if (containsSecret(encoded) || PROOF.test(encoded) || secrets.some(secret => encoded.includes(secret))) return fail("response_invalid");
  }

  async function call(method: "GET" | "POST", path: string, payload: unknown, canRetry: boolean, adminOnly=false, started=performance.now()): Promise<unknown> {
    // Snapshot once: retry must not regenerate the request ID, rewrite the objective or mutate its command.
    let body: string | undefined;
    try { body = payload === undefined ? undefined : canonical(payload); } catch { return fail("request_invalid"); }
    if (body !== undefined && Buffer.byteLength(body) > MAX_BODY_BYTES) return fail("request_invalid");
    const bytes = Buffer.from(body ?? "", "utf8");
    const totalAttempts = canRetry ? attempts : 1;
    let submitted = false;
    let earlierOutcomeUnknown = false;
    let identity: string | undefined;
    for (let attempt = 0; attempt < totalAttempts; attempt++) {
      const remaining = timeoutMs - (performance.now() - started);
      if (remaining <= 0) return fail(method === "POST" && submitted ? "outcome_unknown" : "timeout");
      const budget = Math.max(1, Math.floor(remaining / (totalAttempts - attempt)));
      try {
        return await bounded(async signal => {
          let attestation: string;
          try {
            const actor = await options.authenticate();
            if(adminOnly && (actor.channel!=="admin" || actor.assurance!=="mfa"))return fail("authority_refused");
            const currentIdentity = canonical({ ownerId: actor.ownerId, channel: actor.channel, subject: actor.subject, assurance: actor.assurance });
            if (identity !== undefined && currentIdentity !== identity) return fail("authority_refused");
            identity = currentIdentity;
            attestation = signMissionAuthority({ method, path, body: bytes }, actor, authority);
          } catch { return fail("authority_refused"); }
          // Authentication may have completed after its deadline in an adapter ignoring cancellation.
          if (signal.aborted) return fail("timeout");
          submitted = true;
          const response = await options.transport(`${origin}${path}`, {
            method, headers: { accept: "application/json", "content-type": "application/json", "x-mission-authority": attestation },
            ...(body === undefined ? {} : { body }), signal, redirect: "error", credentials: "omit", cache: "no-store",
          });
          if (signal.aborted) return fail("timeout");
          if (response.redirected || (response.status >= 300 && response.status < 400)) return fail("service_refused", response.status);
          if (!response.ok) return fail("service_refused", response.status);
          const data = await jsonResponse(response);
          safe(data);
          return data;
        }, budget);
      } catch (error) {
        const known = error instanceof MissionGatewayError ? error : new MissionGatewayError("transport_unavailable");
        const retryable = ["timeout", "transport_unavailable"].includes(known.code)
          || (known.code === "service_refused" && (known.status === 429 || (known.status ?? 0) >= 500));
        const ambiguous = method === "POST" && submitted && (["timeout", "transport_unavailable", "response_invalid"].includes(known.code)
          || (known.code === "service_refused" && (known.status ?? 0) >= 500));
        if (retryable && attempt + 1 < totalAttempts) { earlierOutcomeUnknown ||= ambiguous; continue; }
        if (earlierOutcomeUnknown || ambiguous) return fail("outcome_unknown", known.status);
        throw known;
      }
    }
    return fail("transport_unavailable");
  }
  function one(value: unknown): GatewayMissionResponse {
    if (!object(value) || (value.replay !== undefined && typeof value.replay !== "boolean")) return fail("response_invalid");
    return { mission: readMission(value.mission), ...(value.replay === undefined ? {} : { replay: value.replay }) };
  }
  async function postOne(path: string, body: unknown, canRetry = false): Promise<GatewayMissionResponse> {
    const data = await call("POST", path, body, canRetry);
    // A successful HTTP response with an unreadable mission does not prove the write failed.
    try { return one(data); } catch { return fail("outcome_unknown"); }
  }
  async function postProjection<T>(path: string, body: unknown, parse: (value: unknown) => T, canRetry = false, adminOnly = false, started=performance.now()): Promise<T> {
    const data = await call("POST", path, body, canRetry, adminOnly, started);
    // Never repeat a confirmation because its successful reply was incomplete or inconsistent.
    try { return parse(data); } catch { return fail("outcome_unknown"); }
  }
  async function listAccess(started=performance.now()): Promise<GatewayAccessView> {
    const data=await call("GET",`${BASE}/access`,undefined,true,true,started);
    try{return readGatewayAccess(data,authority.ownerId);}catch{return fail("response_invalid");}
  }
  return {
    /** Current scopes and revocable history. Private founder session only, never Telegram. */
    async listAccess(): Promise<GatewayAccessView> { return listAccess(); },
    async issueAccess(input: GatewayAccessIssue): Promise<{grant: ReadGrant}> {
      const started=performance.now();
      let body: GatewayAccessIssue;
      try{body=readGatewayAccessIssue(input);}catch{return fail("request_invalid");}
      // The supplied digest must still describe the current server-owned resource.
      // Snapshot before awaiting: a caller cannot swap the requested scope mid-flight.
      const current=await listAccess(started);
      if(current.scopes[body.capabilityId]?.scopeHash!==body.scopeHash)return fail("request_invalid");
      return postProjection(`${BASE}/access`,body,value=>{
        if(!exactKeys(value,"grant"))return fail("response_invalid");
        const grant=readGatewayAccessGrant(value.grant,authority.ownerId);
        if(grant.capabilityId!==body.capabilityId || grant.scopeHash!==body.scopeHash
          || grant.issuedBy!==authority.gateways.admin?.subject || grant.revokedAt!==null
          || Math.abs(Date.parse(grant.expiresAt)-Date.parse(grant.issuedAt)-body.durationMinutes*60_000)>1)return fail("response_invalid");
        return {grant};
      },false,true,started);
    },
    async revokeAccess(grantId: string): Promise<{revoked: boolean}> {
      const id=requestField(grantId,INTENT_ID);
      // Revocation does not depend on a current scope or on the ability to issue a new grant.
      return postProjection(`${BASE}/access/${id}/revoke`,{},value=>{
        if(!exactKeys(value,"revoked") || typeof value.revoked!=="boolean")return fail("response_invalid");
        return {revoked:value.revoked};
      },false,true);
    },
    async create(input: { requestId: string; objective: string }): Promise<GatewayMissionResponse> {
      if (!object(input) || Object.keys(input).sort().join(",") !== "objective,requestId"
        || typeof input.requestId !== "string" || !REQUEST_ID.test(input.requestId)) return fail("request_invalid");
      try { cleanObjective(input.objective); } catch { return fail("request_invalid"); }
      if (PROOF.test(input.objective) || secrets.some(secret => input.objective.includes(secret))) return fail("request_invalid");
      return postOne(BASE, { requestId: input.requestId, objective: input.objective }, true);
    },
    async list(): Promise<GatewayMissionList> {
      const data = await call("GET", BASE, undefined, true);
      if (!object(data) || !Array.isArray(data.missions) || data.missions.length > 50
        || !object(data.coverage) || !Number.isSafeInteger(data.coverage.limit) || Number(data.coverage.limit) < 1
        || Number(data.coverage.limit) > 50 || typeof data.coverage.exhaustive !== "boolean") return fail("response_invalid");
      return { missions: data.missions.map(readMission), coverage: { limit: Number(data.coverage.limit), exhaustive: data.coverage.exhaustive } };
    },
    async get(id: string): Promise<GatewayMissionResponse> {
      return one(await call("GET", `${BASE}/${missionId(id)}`, undefined, true));
    },
    async telegramDelivery(id: string): Promise<{delivery: TelegramDeliveryStatus | null}> {
      const data = await call("GET", `${BASE}/${missionId(id)}/telegram-delivery`, undefined, true);
      if (!exactKeys(data, "delivery")) return fail("response_invalid");
      try { return {delivery: readTelegramDeliveryStatus(data.delivery)}; }
      catch { return fail("response_invalid"); }
    },
    /** Private source code. Never send this projection to Telegram or execute its contents. */
    async codeReview(id: string, planVersion: number, stepId: string): Promise<{review: CodeReview|null}> {
      const target=missionId(id),step=requestField(stepId,STEP_ID);
      if(!positive(planVersion) || planVersion>999999999)return fail("request_invalid");
      const data=await call("GET",`${BASE}/${target}/plans/${planVersion}/steps/${step}/code-review`,undefined,true,true);
      if(!exactKeys(data,"review"))return fail("response_invalid");
      try{return {review:readCodeReview(data.review,{ownerId:authority.ownerId,missionId:target,stepId:step,planVersion})};}
      catch{return fail("response_invalid");}
    },
    async command(id: string, command: GatewayMissionCommand): Promise<GatewayMissionResponse> {
      return postOne(`${BASE}/${missionId(id)}/commands`, commandBody(command));
    },
    /** Renew only a server-made read/prepare mandate. Recheck remains a separate correlated command. */
    async renew(id: string, planVersion: number): Promise<GatewayMissionResponse> {
      if (!Number.isSafeInteger(planVersion) || planVersion < 1) return fail("request_invalid");
      return postOne(`${BASE}/${missionId(id)}/renew`, { planVersion });
    },
    /** Creates a separate preparation from an existing draft; it grants no execution authority. */
    async createActionMission(parentId: string, input: { requestId: string; sourceStepId: string }): Promise<GatewayMissionResponse> {
      const parent = missionId(parentId);
      if (!exactKeys(input, "requestId,sourceStepId")) return fail("request_invalid");
      const requestId = requestField(input.requestId, /^[A-Za-z0-9][A-Za-z0-9_:.-]{0,159}$/);
      const sourceStepId = requestField(input.sourceStepId, STEP_ID);
      return postProjection(`${BASE}/${parent}/action-missions`, { requestId, sourceStepId }, value => {
        if (!exactKeys(value, "mission,replay") || typeof value.replay !== "boolean"
          || !exactKeys(value.mission, "control,createdAt,id,mode,objective,planVersion,status,steps,updatedAt")) return fail("response_invalid");
        const m = readMission(value.mission);
        if (m.mode !== "prepare" || !m.objective.trim() || m.objective.length > 4_000 || !positive(m.planVersion)
          || !date(m.createdAt) || !date(m.updatedAt)
          || m.steps.some(s => !exactKeys(s, "capability,error,id,needs,result,status"))) return fail("response_invalid");
        return { mission: m, replay: value.replay };
      }, true);
    },
    /** Admin-only on the brain. History is rejected above its bound, never silently truncated. */
    async listActions(id: string): Promise<GatewayActionList> {
      const target = missionId(id);
      const data = await call("GET", `${BASE}/${target}/actions`, undefined, true);
      if (!exactKeys(data, "actions") || !Array.isArray(data.actions) || data.actions.length > MAX_ACTIONS) return fail("response_invalid");
      const actions = data.actions.map(value => readAction(value, { missionId: target }));
      if (new Set(actions.map(action => action.id)).size !== actions.length) return fail("response_invalid");
      return { actions };
    },
    async previewAction(id: string, stepId: string): Promise<GatewayActionResponse> {
      const target = missionId(id), step = requestField(stepId, STEP_ID);
      return postProjection(`${BASE}/${target}/actions/preview`, { stepId: step }, value => {
        if (!exactKeys(value, "action")) return fail("response_invalid");
        return { action: readAction(value.action, { missionId: target, stepId: step }) };
      });
    },
    /** The brain verifies current founder/MFA. No actor, date or approval proof is supplied here. */
    async approveAction(intentId: string, manifestHash: string): Promise<GatewayActionApprovalResponse> {
      const id = requestField(intentId, INTENT_ID), digest = requestField(manifestHash, DIGEST);
      return postProjection(`${BASE}/actions/${id}/approve`, { manifestHash: digest }, value => {
        if (!exactKeys(value, "action,execution") || value.execution !== "not_confirmed") return fail("response_invalid");
        const action = readAction(value.action, { id, manifestHash: digest });
        if (!action.approval) return fail("response_invalid");
        return { action, execution: "not_confirmed" };
      });
    },
    async revokeAction(intentId: string): Promise<{ revoked: boolean }> {
      const id = requestField(intentId, INTENT_ID);
      return postProjection(`${BASE}/actions/${id}/revoke`, {}, value => {
        if (!exactKeys(value, "revoked") || typeof value.revoked !== "boolean") return fail("response_invalid");
        return { revoked: value.revoked };
      });
    },
  };
}

export interface VerifiedTelegramContext {
  senderId: number;
  chatId: number;
  chatType: "private";
  /** Time of real receiver provenance/author verification, not copied from user JSON. */
  authenticatedAt: string;
}
/** The callback is a trusted receiver dependency, never a client-supplied `verified:true`. */
export function telegramActorResolver(options: {
  ownerId: string; telegramSubject: string;
  verifyContext: () => Promise<VerifiedTelegramContext>;
}): () => Promise<AuthenticatedActor> {
  return async () => {
    let context: VerifiedTelegramContext;
    try { context = await options.verifyContext(); } catch { return fail("authority_refused"); }
    if (!context || !Number.isSafeInteger(context.senderId) || context.senderId <= 0
      || !Number.isSafeInteger(context.chatId) || context.chatId !== context.senderId || context.chatType !== "private"
      || String(context.senderId) !== options.telegramSubject) return fail("authority_refused");
    return { ownerId: options.ownerId, channel: "telegram", subject: String(context.senderId),
      assurance: "owner", authenticatedAt: context.authenticatedAt };
  };
}

/** Short plain text only. Never follows a need URL or turns its description into a command. */
export function renderMissionFrench(value: unknown): string {
  let m: GatewayMissionView;
  try { m = readMission(value); } catch { return "Je n’ai pas reçu un état fiable de la mission."; }
  const uncertain = m.steps.some(s => s.status === "uncertain");
  const executing = m.steps.some(s => s.status === "executing");
  if (uncertain) return `${m.control === "cancel" ? "Annulation demandée." : m.control === "pause" ? "Mission en pause." : "L’effet de l’action reste à vérifier."}\nJe ne relance pas cette action sans retrouver son résultat.`;
  if (m.status === "succeeded") {
    const summaries = m.steps.map(s => {
      const result = s.result as JsonObject;
      const summary = String(result.summary).replace(/\s+/g, " ").trim();
      // Renderer is text only; credentials and supplier links never become a suggested action.
      if (containsSecret(summary) || PROOF.test(summary) || /https?:\/\/|\b[A-Za-z0-9_-]{40,}\b/i.test(summary)) return "Résumé à consulter dans l’espace protégé.";
      return summary.length > 230 ? `${summary.slice(0, 229)}…` : summary;
    });
    const draft = m.steps.some(s => object(s.result) && Array.isArray(s.result.evidence)
      && s.result.evidence.some(e => object(e) && e.kind === "draft"));
    return ["Les étapes prévues sont terminées.", ...(m.mode === "test" ? ["Il s’agit d’une simulation."] : []),
      ...(m.control === "cancel" ? ["L’annulation est arrivée après leur fin."] : []),
      ...(draft ? ["Brouillon préparé, à relire."] : []), ...summaries].join("\n");
  }
  if (m.control === "cancel") return executing ? "Annulation demandée. L’action commencée doit encore être vérifiée." : "Mission annulée. Aucune nouvelle étape ne démarre.";
  if (m.status === "cancelled") return "Mission annulée.";
  if (m.control === "pause") return executing ? "Mission en pause. Une étape commencée peut encore se terminer." : "Mission en pause. Aucune nouvelle étape ne démarre.";
  if (m.status === "failed") return "La mission a échoué. Le résultat attendu n’est pas confirmé.";
  if (m.status === "running") return "Ajnaya travaille sur la mission. Le résultat n’est pas encore confirmé.";
  if (m.status === "waiting") {
    const kinds = m.steps.flatMap(s => Array.isArray(s.needs) ? s.needs.flatMap(n => object(n) ? [n.kind] : []) : []);
    if (kinds.includes("verification")) return "La mission attend une vérification. Aucun résultat n’est encore annoncé.";
    if (kinds.includes("access")) return "La mission attend une connexion autorisée. Elle reste conservée.";
    if (kinds.includes("approval")) return "La mission attend ton accord dans l’espace protégé.";
    if (kinds.includes("capability")) return "La capacité nécessaire n’est pas encore disponible. La mission reste en attente.";
    return "La mission attend une information ou une vérification avant de continuer.";
  }
  return "Mission enregistrée. Elle attend son tour.";
}

export function renderGatewayErrorFrench(error: unknown): string {
  if (!(error instanceof MissionGatewayError)) return "Je ne dispose pas d’un résultat fiable. Réessaie depuis le suivi de la mission.";
  if (error.code === "outcome_unknown") return "La réponse s’est interrompue. La demande a pu être enregistrée. Vérifie son suivi avant de la recréer.";
  if (error.code === "authority_refused") return "Ton accès doit être vérifié dans l’espace protégé avant de continuer.";
  if (error.code === "configuration" || error.status === 503) return "Le passage vers les missions n’est pas encore disponible.";
  if (error.code === "request_invalid") return "La demande de mission doit être précisée ou corrigée.";
  return "Le suivi de la mission est temporairement indisponible. Aucun résultat n’est annoncé.";
}
